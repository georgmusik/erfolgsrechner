const SHEET_NAME = 'Daten'; const SETTINGS_SHEET = 'Einstellungen'; function doGet(e) { const result = handleRequest(e); return result; } function doPost(e) { const result = handleRequest(e); return result; } function handleRequest(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (e.postData) { const payload = JSON.parse(e.postData.contents); if (payload.action === 'save') { saveData(ss, payload); return jsonResponse({success: true}); } } const data = loadData(ss); return jsonResponse(data); } catch(err) { return jsonResponse({error: err.toString()}); } finally { lock.releaseLock(); } } function loadData(ss) { let settings = {eventName:'', goalDate:'', cd:120, ipVal:100}; try { const sSheet = ss.getSheetByName(SETTINGS_SHEET); if (sSheet) { settings = { eventName: sSheet.getRange('B1').getValue() || '', goalDate: sSheet.getRange('B2').getValue() || '', cd: parseInt(sSheet.getRange('B3').getValue()) || 120, ipVal: parseInt(sSheet.getRange('B4').getValue()) || 100 }; } } catch(e) {} let members = []; try { const dSheet = ss.getSheetByName(SHEET_NAME); if (dSheet) { const lastRow = dSheet.getLastRow(); if (lastRow > 1) { const data = dSheet.getRange(2, 1, lastRow-1, 4).getValues(); members = data .filter(row => row[0] && row[0].toString().trim() !== '') .map(row => ({ name: row[0].toString().trim(), conv: parseInt(row[1]) || 1, team: row[2].toString().trim() || 'Team 1', active: row[3].toString().toLowerCase() !== 'false' })); } } } catch(e) {} return {settings, members}; } function saveData(ss, payload) { let sSheet = ss.getSheetByName(SETTINGS_SHEET); if (!sSheet) { sSheet = ss.insertSheet(SETTINGS_SHEET); sSheet.getRange('A1:A4').setValues([['Event'],['Zieldatum'],['Zeitraum'],['IP-Wert']]); } const s = payload.settings; sSheet.getRange('B1:B4').setValues([ [s.eventName || ''], [s.goalDate || ''], [s.cd || 120], [s.ipVal || 100] ]); let dSheet = ss.getSheetByName(SHEET_NAME); if (!dSheet) { dSheet = ss.insertSheet(SHEET_NAME); dSheet.getRange('A1:D1').setValues([['Name','Gespräche','Team','Aktiv']]); dSheet.getRange('A1:D1').setFontWeight('bold'); } const members = payload.members || []; const lastRow = dSheet.getLastRow(); if (lastRow > 1) { dSheet.getRange(2, 1, lastRow-1, 4).clearContent(); } if (members.length > 0) { const rows = members.map(m => [ m.name || '', m.conv || 1, m.team || 'Team 1', m.active !== false ? 'true' : 'false' ]); dSheet.getRange(2, 1, rows.length, 4).setValues(rows); } } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
